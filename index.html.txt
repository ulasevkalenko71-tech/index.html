            const tYear = tomorrow.getFullYear();
            const tMonth = String(tomorrow.getMonth() + 1).padStart(2, '0');
            const tDay = String(tomorrow.getDate()).padStart(2, '0');
            const tHours = String(tomorrow.getHours()).padStart(2, '0');
            const tMinutes = String(tomorrow.getMinutes()).padStart(2, '0');
            dateInput.value = `${tYear}-${tMonth}-${tDay}T${tHours}:${tMinutes}`;
        }
    })();
